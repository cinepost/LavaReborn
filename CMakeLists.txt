cmake_minimum_required(VERSION 3.20)
project(HdLava LANGUAGES CXX)
if(POLICY CMP0144)
    cmake_policy(SET CMP0144 NEW)
endif()
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(CMAKE_CXX_STANDARD 17)
set(HOUDINI_ROOT "/opt/hfs20.5" CACHE PATH "Houdini installation")
set(LAVA_RUST_LIBRARY "${CMAKE_SOURCE_DIR}/target/release/liblava_reborn.so" CACHE FILEPATH "Built Rust cdylib")
find_package(Houdini REQUIRED PATHS "${HOUDINI_ROOT}/toolkit/cmake")
file(GLOB HOUDINI_PYTHON_LIB "${HOUDINI_ROOT}/python/lib/libpython3*.so" "${HOUDINI_ROOT}/dsolib/libhboost_python*.so")
list(FILTER HOUDINI_PYTHON_LIB EXCLUDE REGEX "-mt-d-")
add_library(hdLava SHARED hydra/plugin.cpp hydra/interop.cpp)
target_include_directories(hdLava PRIVATE include "${HOUDINI_ROOT}/toolkit/include")
target_link_libraries(hdLava PRIVATE GL Houdini "${LAVA_RUST_LIBRARY}" ${HOUDINI_PYTHON_LIB})
foreach(component hd hf tf gf vt sdf plug arch work trace cameraUtil hgi hgiGL garch usd usdGeom usdImaging usdShade sdr ndr)
    target_link_libraries(hdLava PRIVATE "${HOUDINI_ROOT}/dsolib/libpxr_${component}.so")
endforeach()
set_target_properties(hdLava PROPERTIES BUILD_RPATH "${HOUDINI_ROOT}/dsolib;${CMAKE_SOURCE_DIR}/target/release" INSTALL_RPATH "$ORIGIN")
foreach(resource openpbr_defs.usda lava_openpbr.glslfx)
    configure_file("hydra/${resource}" "${CMAKE_BINARY_DIR}/resources/${resource}" COPYONLY)
    install(FILES "hydra/${resource}" DESTINATION hdLava/resources)
endforeach()
configure_file(hydra/plugInfo.json.in "${CMAKE_BINARY_DIR}/resources/plugInfo.json" COPYONLY)
install(TARGETS hdLava LIBRARY DESTINATION hdLava)
install(FILES "${LAVA_RUST_LIBRARY}" DESTINATION hdLava)
install(FILES "${CMAKE_BINARY_DIR}/resources/plugInfo.json" DESTINATION hdLava/resources)
install(FILES hydra/UsdRenderers.json DESTINATION houdini)
option(LAVA_BUILD_HYDRA_SMOKE "Build direct Hydra integration test" ON)
if(LAVA_BUILD_HYDRA_SMOKE)
    add_executable(lava_hydra_smoke tests/hydra_smoke.cpp)
    target_include_directories(lava_hydra_smoke PRIVATE include "${HOUDINI_ROOT}/toolkit/include")
    target_link_libraries(lava_hydra_smoke PRIVATE Houdini "${LAVA_RUST_LIBRARY}" ${HOUDINI_PYTHON_LIB})
    foreach(component hd hf tf gf vt sdf plug arch work trace cameraUtil)
        target_link_libraries(lava_hydra_smoke PRIVATE "${HOUDINI_ROOT}/dsolib/libpxr_${component}.so")
    endforeach()
endif()
# Houdini initializes USD discovery through HOUDINI_PATH/dso/usd_plugins.
# PXR_PLUGINPATH_NAME from houdini.env can be applied after USD initialization.
install(FILES hydra/houdiniPlugInfo.json DESTINATION houdini/dso/usd_plugins RENAME plugInfo.json)

if(LAVA_BUILD_HYDRA_SMOKE)
    add_executable(lava_usd_smoke tests/usd_smoke.cpp tests/gl_context.cpp)
    target_include_directories(lava_usd_smoke PRIVATE include "${HOUDINI_ROOT}/toolkit/include")
    target_compile_definitions(lava_usd_smoke PRIVATE LAVA_HOUDINI_PLUGIN_PATH="${HOUDINI_ROOT}/houdini/dso/usd_plugins")
    target_link_libraries(lava_usd_smoke PRIVATE "${LAVA_RUST_LIBRARY}" GL X11 Houdini ${HOUDINI_PYTHON_LIB})
    foreach(component hd hdx hf tf gf vt sdf plug arch work trace cameraUtil usd usdGeom usdImaging hgi garch)
        target_link_libraries(lava_usd_smoke PRIVATE "${HOUDINI_ROOT}/dsolib/libpxr_${component}.so")
    endforeach()
endif()
if(LAVA_BUILD_HYDRA_SMOKE)
    add_library(lava_houdini_smoke SHARED tests/houdini_smoke.cpp tests/gl_context.cpp)
    target_include_directories(lava_houdini_smoke PRIVATE include "${HOUDINI_ROOT}/toolkit/include")
    target_compile_definitions(lava_houdini_smoke PRIVATE LAVA_HOUDINI_PLUGIN_PATH="${HOUDINI_ROOT}/houdini/dso/usd_plugins")
    target_link_libraries(lava_houdini_smoke PRIVATE "${LAVA_RUST_LIBRARY}" GL X11 Houdini ${HOUDINI_PYTHON_LIB})
    foreach(component hd hf tf gf vt sdf plug arch work trace usd usdGeom garch)
        target_link_libraries(lava_houdini_smoke PRIVATE "${HOUDINI_ROOT}/dsolib/libpxr_${component}.so")
    endforeach()
endif()

# Digital assets are generated with the matching Houdini Python API.
add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/houdini/otls/lava_render_settings.hda" "${CMAKE_BINARY_DIR}/houdini/otls/lava_light_properties.hda"
    COMMAND "${HOUDINI_ROOT}/bin/hython" "${CMAKE_SOURCE_DIR}/scripts/build_houdini_assets.py" "${CMAKE_BINARY_DIR}/houdini/otls"
    DEPENDS scripts/build_houdini_assets.py scripts/expose_hip_settings.py
    COMMENT "Building Lava Solaris digital assets")
add_custom_target(lava_houdini_assets ALL DEPENDS "${CMAKE_BINARY_DIR}/houdini/otls/lava_render_settings.hda" "${CMAKE_BINARY_DIR}/houdini/otls/lava_light_properties.hda")
install(FILES "${CMAKE_BINARY_DIR}/houdini/otls/lava_render_settings.hda" "${CMAKE_BINARY_DIR}/houdini/otls/lava_light_properties.hda" DESTINATION houdini/otls)

# Numeric scene serialization regression; no Houdini or GPU dependencies.
add_executable(lava_json_writer_smoke tests/json_writer_smoke.cpp)
target_compile_options(lava_json_writer_smoke PRIVATE -UNDEBUG)
